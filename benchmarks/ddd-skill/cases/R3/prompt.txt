`src/catalog/domain/catalog.aggregate.ts`
```ts
export class Catalog extends AggregateRoot<CatalogId, CatalogEvent> {
  private products: Product[] = [];
  private categories: Category[] = [];
  private suppliers: SupplierEntry[] = [];
  private priceLists: PriceList[] = [];
  private promotions: Promotion[] = [];
  private reviews: Review[] = [];

  addProduct(p: Product): void { this.products.push(p); }
  removeProduct(id: string): void { this.products = this.products.filter(p => p.id !== id); }
  addCategory(c: Category): void { this.categories.push(c); }
  removeCategory(c: Category): void { this.categories = this.categories.filter(x => x !== c); }
  registerSupplier(s: SupplierEntry): void { this.suppliers.push(s); }
  publishPriceList(l: PriceList): void { this.priceLists.push(l); }
  retirePriceList(l: PriceList): void { this.priceLists = this.priceLists.filter(x => x !== l); }
  startPromotion(p: Promotion): void { this.promotions.push(p); }
  endPromotion(p: Promotion): void { this.promotions = this.promotions.filter(x => x !== p); }
  addReview(r: Review): void { this.reviews.push(r); }
  moderateReview(r: Review): void { r.hide(); }
  recalculateAllPrices(): void { this.products.forEach(p => p.reprice(this.priceLists)); }
}
```
