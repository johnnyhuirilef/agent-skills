```ts
function ensureValidEmail(value: string): string {
  const normalized = value.trim().toLowerCase();
  if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(normalized)) {
    throw new DomainValidationError(`Invalid email: ${value}`);
  }
  return normalized;
}

export class Email extends ValueObject<string> {
  constructor(value: string) {
    super(ensureValidEmail(value));
  }
}
```

`src/customer/domain/model/customer.aggregate.ts`
```ts
export enum CustomerStatus { ACTIVE = 'active', DEACTIVATED = 'deactivated' }
export type CustomerSnapshot = { id: string; email: string; status: CustomerStatus };

export class Customer extends AggregateRoot<CustomerId, CustomerEvent> {
  private constructor(id: CustomerId, private email: Email, private status: CustomerStatus) {
    super(id);
  }

  static create(id: CustomerId, email: Email, deps: DomainDeps = systemDeps): Customer {
    const customer = new Customer(id, email, CustomerStatus.ACTIVE);
    const now = deps.clock.now();
    customer.addDomainEvent(
      new CustomerRegistered({ customerId: id.value, email: email.value, registeredAt: now.toISOString() }, now),
    );
    return customer;
  }

  static restore(snapshot: CustomerSnapshot): Customer {
    return new Customer(new CustomerId(snapshot.id), new Email(snapshot.email), snapshot.status);
  }

  changeEmail(email: Email): void {
    this.ensureActive();
    this.email = email;
  }

  deactivate(deps: DomainDeps = systemDeps): void {
    this.ensureActive();
    this.status = CustomerStatus.DEACTIVATED;
    const now = deps.clock.now();
    this.addDomainEvent(new CustomerDeactivated({ customerId: this.id.value, deactivatedAt: now.toISOString() }, now));
  }

  toPrimitives(): CustomerSnapshot {
    return { id: this.id.value, email: this.email.value, status: this.status };
  }

  private ensureActive(): void {
    if (this.status === CustomerStatus.DEACTIVATED) throw new CustomerDeactivatedError(this.id.value);
  }
}
```

`src/customer/domain/events/customer.events.ts`
```ts
type CustomerRegisteredPayload = { readonly customerId: string; readonly email: string; readonly registeredAt: string };
type CustomerDeactivatedPayload = { readonly customerId: string; readonly deactivatedAt: string };

export class CustomerRegistered extends DomainEvent<CustomerRegisteredPayload> {
  static readonly EVENT_NAME = 'customer.customer-registered.v1';
  readonly eventName = CustomerRegistered.EVENT_NAME;

  static fromPrimitives(data: EventPrimitives<CustomerRegisteredPayload>): CustomerRegistered {
    return new CustomerRegistered(data.payload, new Date(data.occurredAt));
  }
}

export class CustomerDeactivated extends DomainEvent<CustomerDeactivatedPayload> {
  static readonly EVENT_NAME = 'customer.customer-deactivated.v1';
  readonly eventName = CustomerDeactivated.EVENT_NAME;

  static fromPrimitives(data: EventPrimitives<CustomerDeactivatedPayload>): CustomerDeactivated {
    return new CustomerDeactivated(data.payload, new Date(data.occurredAt));
  }
}

export type CustomerEvent = CustomerRegistered | CustomerDeactivated;
```

`src/customer/application/command/deactivate-customer.handler.ts`
```ts
export class DeactivateCustomerHandler {
  constructor(private readonly repo: CustomerRepositoryPort, private readonly eventBus: EventBusPort) {}

  async execute(cmd: { customerId: string }): Promise<void> {
    const customer = await this.repo.findById(new CustomerId(cmd.customerId));
    customer.deactivate();
    await this.repo.save(customer);
    await this.eventBus.publish(customer.pullDomainEvents());
  }
}
```
