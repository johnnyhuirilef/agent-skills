`src/shoppingAndInvoice/domain/invoice.entity.ts`
```ts
import { InvoiceTypeOrmRepository } from '../infrastructure/invoice.typeorm.repository';
import { formatInvoiceNumber } from '../../utils/billing-helpers';

export class Invoice extends Entity<InvoiceId> {
  private paid = false;
  private readonly repo = new InvoiceTypeOrmRepository();

  constructor(id: InvoiceId, private readonly seq: number) {
    super(id);
  }

  async markAsPaid(): Promise<void> {
    if (this.paid) throw new InvoiceAlreadyPaidError(this.id.toPrimitives());
    this.paid = true;
    await this.repo.save(this);
  }

  number(): string {
    return formatInvoiceNumber(this.seq);
  }
}
```

`src/shoppingAndInvoice/domain/tax-calculation.service.ts`
```ts
import axios from 'axios';

export class TaxCalculationService {
  async calculate(country: string, net: Money): Promise<Money> {
    const res = await axios.get(`https://tax-api.example.com/rates/${country}`);
    return net.multiply(res.data.rate);
  }
}
```
